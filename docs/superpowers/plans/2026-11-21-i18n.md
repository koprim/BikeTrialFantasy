# Internationalisation (fr, es, en) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. The project skill `.claude/skills/solid-clean-code` is MANDATORY for every task.

**Goal:** The game shows in French, Spanish or English (interface and game texts), detects the language, lets the player change it, and an Infos page shows how much of each language is translated.

**Architecture:** One language registry (`src/i18n/languages.js`) feeds i18next (interface texts, `ui` namespace), a `gameText()` reader (card / season / round texts by id, with a fallback chain), the progress calculator, the SQL seed builder and the report script. The engine stops producing sentences: it returns codes (`validateCardTeam`), raw values (`formatValue`) and ids; screens translate. Seed JSON keeps only mechanics; French texts live in `src/i18n/locales/fr/game.json`; Supabase stores texts as `jsonb { "<code>": text }`.

**Tech Stack:** Expo SDK 58, React 19.3, React Native 0.88, i18next + react-i18next, expo-localization, AsyncStorage, Jest 30 (jest-expo), Supabase (Postgres migrations, seed.sql).

**Spec:** `docs/superpowers/specs/2026-11-20-i18n-design.md`

**Conventions used by every task:**
- `src/i18n/languages.js`, `game.js`, `progress.js` are **CommonJS** (`require` / `module.exports`) so plain-node scripts (`build-sql-seed`, `i18n-report`) can load them. `index.js`, `detect.js`, `useGameText.js`, `LanguageGate.js` are ES modules (they import React Native packages).
- Components translate with `const { t } = useTranslation();` (`react-i18next`). Non-component helpers (`errorMessages.js`, `eventLabel.js`, `teamErrors.js`) use the singleton: `import i18n from '../i18n'; i18n.t(...)`.
- Interface keys live in `src/i18n/locales/<code>/ui.json` (one namespace `ui`, the default). Interpolation is `{{name}}`. Keep the French text **exactly** as it is today so existing screen tests keep passing.
- Game texts are read with `useGameText()` → `{ cardName(id), effectText(cardId, index, value), seasonName(id), eventName(id) }`.
- No file other than `languages.js` names a language code, except `test/i18nSetup.js` via `REFERENCE_LANGUAGE`.

```js
import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { LANGUAGES, REFERENCE_LANGUAGE, FALLBACK_LANGUAGE, languageByCode } from './languages';
import { detectLanguage, LANGUAGE_KEY } from './detect';

i18n.use(initReactI18next).init({
  resources: Object.fromEntries(LANGUAGES.map((language) => [language.code, { ui: language.ui }])),
  lng: REFERENCE_LANGUAGE, // replaced by startLanguage() before the first screen renders
  fallbackLng: [FALLBACK_LANGUAGE, REFERENCE_LANGUAGE],
  ns: ['ui'],
  defaultNS: 'ui',
  returnEmptyString: false, // an empty draft text counts as missing
  interpolation: { escapeValue: false }, // React already escapes
  initAsync: false, // i18next < 24: initImmediate: false
});

/** Applies the starting language (saved choice, device, fallback). Never rejects. */
export async function startLanguage(deps) {
  try {
    await i18n.changeLanguage(await detectLanguage(deps));
  } catch {
    // keep the current language
  }
  return i18n.language;
}

/** Applies a registry language now and remembers it on this device. */
export async function setLanguage(code, storage = AsyncStorage) {
  if (!languageByCode(code)) return;
  await i18n.changeLanguage(code);
  try {
    await storage.setItem(LANGUAGE_KEY, code);
  } catch {
    // the choice still applies for this session
  }
}

export default i18n;
```

```js
import AsyncStorage from '@react-native-async-storage/async-storage';
import { getLocales } from 'expo-localization';
import { FALLBACK_LANGUAGE, languageByCode } from './languages';

export const LANGUAGE_KEY = 'language';

const known = (code) => !!code && !!languageByCode(code);
const baseCode = (locale) => String(locale.languageCode || locale.languageTag || '').split('-')[0].toLowerCase();

async function savedLanguage(storage) {
  try {
    return await storage.getItem(LANGUAGE_KEY);
  } catch {
    return null;
  }
}

function deviceLanguage(deviceLocales) {
  try {
    return (deviceLocales() || []).map(baseCode).find(known) || null;
  } catch {
    return null;
  }
}

/** Saved choice if declared in the registry, else first device language in the registry, else the fallback. */
export async function detectLanguage({ storage = AsyncStorage, deviceLocales = getLocales } = {}) {
  const saved = await savedLanguage(storage);
  if (known(saved)) return saved;
  return deviceLanguage(deviceLocales) || FALLBACK_LANGUAGE;
}
```

```js
const fs = require('fs');
const path = require('path');

const FRENCH_GAME_TEXTS = path.join(__dirname, '..', '..', 'src', 'i18n', 'locales', 'fr', 'game.json');

// Seed effects use {value}; locale files use the i18next form {{value}}.
const toTemplate = (description) => description.split('{value}').join('{{value}}');

function splitSeason({ name, events, ...season }, texts) {
  texts.seasons[season.id] = { name };
  return {
    ...season,
    events: events.map(({ name: eventName, ...event }) => {
      texts.events[event.id] = { name: eventName };
      return event;
    }),
  };
}

function splitCard({ name, effects, ...card }, texts) {
  texts.cards[card.id] = { name, effects: effects.map((effect) => toTemplate(effect.description)) };
  return { ...card, effects: effects.map(({ description, ...effect }) => effect) };
}

/** Generated data in, { seasons, cards } mechanics + French texts out. `cards` is optional. */
function splitGameTexts({ seasons, cards }) {
  const texts = { cards: {}, seasons: {}, events: {} };
  return {
    seasons: seasons.map((season) => splitSeason(season, texts)),
    cards: cards ? cards.map((card) => splitCard(card, texts)) : undefined,
    texts,
  };
}

/** Seasons and events are replaced; card texts too, unless the generator produced none. */
function mergeGameTexts(current, texts) {
  const hasCards = Object.keys(texts.cards).length > 0;
  return { cards: hasCards ? texts.cards : current.cards || {}, seasons: texts.seasons, events: texts.events };
}

function writeFrenchGameTexts(texts, file = FRENCH_GAME_TEXTS) {
  const current = fs.existsSync(file) ? JSON.parse(fs.readFileSync(file, 'utf8')) : {};
  fs.writeFileSync(file, `${JSON.stringify(mergeGameTexts(current, texts), null, 2)}\n`);
}

module.exports = { splitGameTexts, mergeGameTexts, writeFrenchGameTexts, FRENCH_GAME_TEXTS };
```

- [ ] **Step 8: Implement** `src/i18n/game.js`

```js
const REGISTRY = require('./languages');

const text = (value) => (typeof value === 'string' && value !== '' ? value : undefined);

// chosen language → FALLBACK_LANGUAGE → REFERENCE_LANGUAGE, each once.
function chainFor(lang, registry) {
  const codes = [...new Set([lang, registry.FALLBACK_LANGUAGE, registry.REFERENCE_LANGUAGE])];
  return codes.map(registry.languageByCode).filter(Boolean).map((language) => language.game || {});
}

/** Game texts (cards, effects, seasons, rounds) in `lang`, with the fallback chain. */
function gameText(lang, registry = REGISTRY) {
  const chain = chainFor(lang, registry);
  const pick = (read) => chain.map(read).map(text).find(Boolean);
  return {
    cardName: (id) => pick((game) => game.cards?.[id]?.name) ?? id,
    effectText: (cardId, index, value) =>
      (pick((game) => game.cards?.[cardId]?.effects?.[index]) ?? '').split('{{value}}').join(value),
    seasonName: (id) => pick((game) => game.seasons?.[id]?.name) ?? id,
    eventName: (id) => pick((game) => game.events?.[id]?.name) ?? id,
  };
}

module.exports = { gameText };
```

```js
function validateCardTeam(team, { cardsById, ridersById, owned, balance }) {
  const slots = team.slots || {};
  const errors = [];
  const fail = (code, slot = null) => errors.push({ code, slot });
  const used = new Set();
  const riderIds = [];
  const checkCard = (ref, type, where) => {
    const card = cardsById[ref.cardId];
    if (!card || card.type !== type) {
      fail('CARD_INVALID', where);
      return null;
    }
    if (!card.rarities.includes(ref.rarity)) {
      fail('RARITY_UNAVAILABLE', where);
      return null;
    }
    if (!ownsCard(owned, ref)) fail('CARD_NOT_OWNED', where);
    const key = `${ref.cardId}:${ref.rarity}`;
    if (used.has(key)) fail('CARD_ALREADY_USED', where);
    used.add(key);
    return card;
  };
  SLOTS.forEach((slot) => {
    const entry = slots[slot.id];
    if (!entry) {
      fail('SLOT_EMPTY', slot.id);
      return;
    }
    const card = checkCard(entry, 'PILOT', slot.id);
    if (!card) return;
    const rider = ridersById[card.subject];
    if (!rider || rider.category !== slot.category) fail('WRONG_CATEGORY', slot.id);
    else if (riderIds.includes(rider.id)) fail('RIDER_ALREADY_IN_TEAM', slot.id);
    else riderIds.push(rider.id);
    const techniques = entry.techniques || [];
    if (techniques.length > balance.slots.techniquesByPilotRarity[entry.rarity]) fail('TOO_MANY_TECHNIQUES', slot.id);
    techniques.forEach((t) => checkCard(t, 'TECHNIQUE', slot.id));
  });
  const equipment = team.equipment || [];
  if (equipment.length > balance.slots.equipment) fail('TOO_MANY_EQUIPMENT');
  equipment.forEach((e) => checkCard(e, 'EQUIPMENT', 'equipment'));
  if (team.referee) checkCard(team.referee, 'REFEREE', 'referee');
  return { valid: errors.length === 0, errors };
}
```

```js
function localized(texts, read) {
  const out = {};
  Object.entries(texts).forEach(([code, game]) => {
    const value = read(game || {});
    if (typeof value === 'string' && value !== '') out[code] = value;

}
```

```js
const REGISTRY = require('./languages');

/** 'a.b.0' style paths of every leaf (string) under an object or array. */
function leafKeys(value, prefix = '') {
  if (value && typeof value === 'object') {
    return Object.entries(value).flatMap(([key, child]) => leafKeys(child, prefix ? `${prefix}.${key}` : key));
  }
  return prefix ? [prefix] : [];
}

const valueAt = (object, path) => path.split('.').reduce((node, key) => (node == null ? undefined : node[key]), object);
const filled = (value) => typeof value === 'string' && value.trim() !== '';

/** Share of the reference keys (ui + game) present and non-empty in `language`. */
function translationProgress(language, reference) {
  const keys = ['ui', 'game'].flatMap((file) => leafKeys(reference[file]).map((key) => [file, key]));
  const missing = keys.filter(([file, key]) => !filled(valueAt(language[file], key))).map(([file, key]) => `${file}:${key}`);
  const percent = keys.length ? Math.floor(((keys.length - missing.length) / keys.length) * 100) : 100;
  return { percent, missing };
}

/** One row per registry language: { code, name, percent, missing }. */
function languagesProgress(registry = REGISTRY) {
  const reference = registry.languageByCode(registry.REFERENCE_LANGUAGE);
  return registry.LANGUAGES.map((language) => ({
    code: language.code,
    name: language.name,
    ...translationProgress(language, reference),
  }));
}

module.exports = { leafKeys, valueAt, translationProgress, languagesProgress };
```
